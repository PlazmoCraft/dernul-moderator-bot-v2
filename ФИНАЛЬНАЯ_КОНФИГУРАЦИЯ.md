### Функция создания круглого аватара
```python
def _draw_circular_avatar(bg, avatar_bytes, position, size):
    """Рисует круглый аватар на фоне
    
    Args:
        bg: Фоновое изображение (PIL Image)
        avatar_bytes: Байты аватара пользователя
        position: Кортеж (x, y) позиции вставки
        size: Диаметр круга (270px)
    """
    # Загружаем и изменяем размер аватара
    avatar = Image.open(io.BytesIO(avatar_bytes)).convert("RGBA")
    avatar = avatar.resize((size, size), Image.Resampling.LANCZOS)
    
    # Создаем круглую маску
    mask = Image.new("L", (size, size), 0)
    draw_mask = ImageDraw.Draw(mask)
    draw_mask.ellipse((0, 0, size, size), fill=255)
    
    # Применяем маску (делаем аватар круглым)
    avatar.putalpha(mask)
    
    # Вставляем аватар на фон с прозрачностью
    bg.paste(avatar, position, avatar)
```

```python
async def create_welcome_image(member: discord.Member):
    # 1. Загружаем шаблон join.png (1748x510)
    bg = Image.open("assets/join.png").convert("RGBA")
    
    # 2. Получаем аватар пользователя (высокое качество 512px)
    avatar_bytes = await member.display_avatar.replace(size=512, format='png').read()
    
    # 3. Рисуем круглый аватар (270x270 на позиции 88, 108)
    _draw_circular_avatar(bg, avatar_bytes, (88, 108), 270)
    
    # 4. Рисуем текст
    draw = ImageDraw.Draw(bg)
    font = _load_minecraft_font(42)  # Minecraft.ttf, 42px
    
    text = f"{member.display_name} зашёл на сервер."
    
    # 5. Авто-уменьшение шрифта если никнейм длинный (38-42px)
    max_width = bg.width - 460 - 50  # От позиции текста до края с отступом
    font = _fit_text_to_width(draw, text, font, max_width, min_size=38)
    
    # 6. Рисуем белый текст на позиции (460, 285)
    draw.text((460, 285), text, fill="#FFFFFF", font=font)
    
    # 7. Сохраняем как PNG
    return discord.File(...)
```

## 🔧 Автоматическое масштабирование шрифта

```python
def _fit_text_to_width(draw, text, font, max_width, min_size=38):
    """Уменьшает размер шрифта пока текст не поместится
    
    Args:
        draw: Объект ImageDraw
        text: Текст для измерения
        font: Начальный шрифт (42px)
        max_width: Максимальная ширина (1748 - 460 - 50 = 1238px)
        min_size: Минимальный размер (38px)
    
    Returns:
        Подогнанный шрифт
    """
    current_size = font.size  # Начинаем с 42
    
    while current_size > min_size:
        test_font = _load_minecraft_font(current_size)
        bbox = draw.textbbox((0, 0), text, font=test_font)
        text_width = bbox[2] - bbox[0]
        
        if text_width <= max_width:
            return test_font  # Текст помещается!
        
        current_size -= 2  # Уменьшаем на 2px
    
    return _load_minecraft_font(min_size)  # Минимум 38px
```
